```{raw} html
<script>
document.addEventListener('DOMContentLoaded', function () {
  var quiz = document.getElementById('xx-quiz');
  var left = document.getElementById('xx-left');
  var run = document.getElementById('xx-run');
  var report = document.getElementById('xx-report');
  var total = 10;

  function answered() {
    return quiz.querySelectorAll('input[type=radio]:checked').length;
  }
  function tick() {
    var n = answered();
    left.textContent = n === total ? '已答完，可以出报告' : ('还有 ' + (total - n) + ' 题未答');
    run.disabled = (n < total);
  }
  quiz.addEventListener('change', tick);

  run.addEventListener('click', function () {
    var D = 0, S = 0, M = 0;
    for (var i = 1; i <= 10; i++) {
      var el = quiz.querySelector('input[name=q' + i + ']:checked');
      if (!el) continue;
      var v = parseInt(el.value, 10);
      var dim = el.closest('.xx-q').getAttribute('data-dim');
      if (dim === 'D') D += v; else if (dim === 'S') S += v; else M += v;
    }
    // 归一化到 0-1（雷达半径）
    var rS = S / 9, rM = M / 9, rD = D / 12;
    // 三角形三点：顶(160,40)=S，右下(270,230)=M，左下(50,230)=外包风险
    var cx = 160, cy = 160;
    function pt(r, tx, ty) { return (cx + (tx - cx) * r).toFixed(1) + ',' + (cy + (ty - cy) * r).toFixed(1); }
    document.getElementById('xx-shape').setAttribute('points',
      pt(rS, 160, 40) + ' ' + pt(rM, 270, 230) + ' ' + pt(rD, 50, 230));
    document.getElementById('xx-bar-D').style.width = (rD * 100) + '%';
    document.getElementById('xx-bar-S').style.width = (rS * 100) + '%';
    document.getElementById('xx-bar-M').style.width = (rM * 100) + '%';
    document.getElementById('xx-val-D').textContent = D + ' / 12';
    document.getElementById('xx-val-S').textContent = S + ' / 9';
    document.getElementById('xx-val-M').textContent = M + ' / 9';

    var level;
    if (D >= 9) level = '高危 · 学习正在被外包';
    else if (D >= 5) level = '警戒 · 流畅陷阱已出现';
    else level = '安全 · 学习主权在自己手里';
    document.getElementById('xx-verdict').innerHTML =
      '<div class="xx-badge">' + level + '</div>' +
      '<p>外包度 ' + D + '/12（越高越危险）；策略 ' + S + '/9、动机 ' + M + '/9（越高越健康）。' +
      '学习的主权在你自己的神经元上，任何外部工具都只是条件——条件再好，建构发生在你这里才算数。</p>';
    report.hidden = false;
    report.scrollIntoView({behavior: 'smooth', block: 'start'});
  });
  tick();
});
</script>
```
